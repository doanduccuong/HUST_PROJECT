\section*{DANH MỤC THUẬT NGỮ VÀ VIẾT TẮT}
\phantomsection \addcontentsline{toc}{section}{\numberline {} DANH MỤC THUẬT NGỮ VÀ VIẾT TẮT}

\begin{tabular}{p{2.5cm} p{13cm}}
    \textbf{AI} & Artificial Intelligence (Trí tuệ nhân tạo) \\
    \textbf{API} & Application Programming Interface (Giao diện lập trình ứng dụng) \\
    \textbf{AUC} & Area Under the ROC Curve (Diện tích dưới đường cong ROC) \\
    \textbf{BBox} & Bounding Box (Khung giới hạn khuôn mặt) \\
    \textbf{BI} & Business Intelligence (Kinh doanh thông minh / Phân tích dữ liệu kinh doanh) \\
    \textbf{CNN} & Convolutional Neural Network (Mạng thần kinh tích chập) \\
    \textbf{CRM} & Customer Relationship Management (Quản trị quan hệ khách hàng) \\
    \textbf{CSAT} & Customer Satisfaction Score (Chỉ số hài lòng khách hàng) \\
    \textbf{CX} & Customer Experience (Trải nghiệm khách hàng) \\
    \textbf{EMA} & Exponential Moving Average (Trung bình trượt hàm mũ) \\
    \textbf{FACS} & Facial Action Coding System (Hệ thống mã hóa hành động cơ mặt) \\
    \textbf{FAR} & False Acceptance Rate (Tỷ lệ chấp nhận sai) \\
    \textbf{FER} & Facial Expression Recognition (Nhận dạng biểu cảm khuôn mặt) \\
    \textbf{FPS} & Frames Per Second (Số khung hình trên giây) \\
    \textbf{FRR} & False Rejection Rate (Tỷ lệ từ chối sai) \\
    \textbf{JSONB} & Binary JSON (Định dạng lưu trữ JSON nhị phân trong CSDL) \\
    \textbf{JWT} & JSON Web Token (Mã định danh xác thực phân quyền) \\
    \textbf{KPI} & Key Performance Indicator (Chỉ số đo lường hiệu quả cốt lõi) \\
    \textbf{NPS} & Net Promoter Score (Chỉ số đo lường mức độ quảng bá) \\
    \textbf{POS} & Point of Sale (Điểm bán hàng / Quầy thu ngân) \\
    \textbf{REST} & Representational State Transfer (Kiến trúc dịch vụ web chuẩn) \\
    \textbf{ROC} & Receiver Operating Characteristic (Đường cong đặc tính hoạt động máy thu) \\
\end{tabular}
